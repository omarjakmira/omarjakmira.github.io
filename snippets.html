Debounce Function

A simple debounce utility to limit function execution rate.

javascript
utilities
performance
async
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}
// Usage// Usage
const handleSearch = debounce((query) => {
  console.log('Searching for:'for:', query);
}, 300);

Deep Clone Object

Recursively clone objects without external dependencies.

javascript
utilities
objects
immutability
function deepClone(obj) {
  ifif (obj === null || typeof obj !== 'object') {
    return obj;
  }
  
  ifif (obj instanceof Date) {
    return new Date(obj.getTime());
  }
  
  ifif (obj instanceof Array) {
    return obj.map(item => deepClone(item));
  }
  
  ifif (obj instanceof Object) {
    const clonedObj = {};
    forfor (const key in obj) {
      ifif (obj.hasOwnProperty(key)) {
        clonedObj[key] = deepClone(obj[key]);
      }
    }
    return clonedObj;
  }
}

Type-Safe Event Emitter

A strongly-typed event emitter pattern in TypeScript.

typescript
patterns
type-safety
events
type EventMap = {
  [key: string]: any;
};
class TypedEventEmitter<T extends EventMap> {
  private listeners: { [K in keyof T]?: Array<(data: T[K]) => void> } = {};
  on<K extends keyof T>(event: K, listener: (data: T[K]) => void): void {
    ifif (!this.listeners[event]) {
      this.listeners[event] = [];
    }
    this.listeners[event]!.push(listener);
  }
  emit<K extends keyof T>(event: K, data: T[K]): void {
    this.listeners[event]?.forEach(listener => listener(data));
  }
}
// Usage// Usage
type MyEvents = {
  userLogin: { userId: string; timestamp: number };
  dataUpdate: { records: number };
};
const emitter = new TypedEventEmitter<MyEvents>();
emitter.on('userLogin', (data) => {
  console.log(`User ${data.userId} logged in`in`);
});

useLocalStorage Hook

React hook for syncing state with localStorage.

typescript
react
hooks
storage
import { useState, useEffect } from 'react';
function useLocalStorage<T>(key: string, initialValue: T) {
  const [storedValue, setStoredValue] = useState<T>(() => {
    try {
      const item = window.localStorage.getItem(key);
      return item ? JSON.parse(item) : initialValue;
    } catchcatch (error) {
      console.error(error);
      return initialValue;
    }
  });
  const setValue = (value: T | ((val: T) => T)) => {
    try {
      const valueToStore = value instanceof Function ? value(storedValue) : value;
      setStoredValue(valueToStore);
      window.localStorage.setItem(key, JSON.stringify(valueToStore));
    } catchcatch (error) {
      console.error(error);
    }
  };
  return [storedValue, setValue] as const;
}
// Usage// Usage
const [theme, setTheme] = useLocalStorage('theme', 'dark');

Retry with Exponential Backoff

Retry failed async operations with exponential backoff.

async function retryWithBackoff(
  fn,
  maxRetries = 3,
  baseDelay = 1000
) {
  forfor (let i = 0; i < maxRetries; i++) {
    try {
      return await fn();
    } catchcatch (error) {
      ifif (i === maxRetries - 1) throw error;
      
      const delay = baseDelay * Math.pow(2, i);
      console.log(`Retry ${i + 1}/${maxRetries} after ${delay}ms`+ 1}/${maxRetries} after ${delay}ms`);
      await new Promise(resolve => setTimeout(resolve, delay));
    }
  }
}

Intersection Observer Hook

React hook for detecting element visibility.

typescript
react
hooks

import { useEffect, useRef, useState } from 'react';
function useIntersectionObserver(
  options?: IntersectionObserverInit
) {
  const [isIntersecting, setIsIntersecting] = useState(false);
  const targetRef = useRef<HTMLDivElement>(null);
  useEffect(() => {
    const target = targetRef.current;
    ifif (!target) return;
    const observer = new IntersectionObserver(([entry]) => {
      setIsIntersecting(entry.isIntersecting);
    }, options);
    observer.observe(target);
    returnreturn () => observer.disconnect();
  }, [options]);
  return { targetRef, isIntersecting };
}
// Usage// Usage
const { targetRef, isIntersecting } = useIntersectionObserver({
  threshold: 0.5
});
